cmake_minimum_required(VERSION 3.16)
project(psp2_gpuprof VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(psp2_gpuprof_client STATIC src/client_trace.c)
target_include_directories(psp2_gpuprof_client PUBLIC include)
target_compile_options(psp2_gpuprof_client PRIVATE -Wall -Wextra -Werror)
add_library(gpuprof_session STATIC src/session.c)
target_include_directories(gpuprof_session PUBLIC include PRIVATE src)
target_compile_options(gpuprof_session PRIVATE -Wall -Wextra -Werror)

if(CMAKE_CROSSCOMPILING)
    include("${VITASDK}/share/vita.cmake" REQUIRED)
    # Kernel contexts require aligned accesses and cannot assume VFP/NEON state.
    target_compile_options(gpuprof_session PRIVATE -mgeneral-regs-only -mno-unaligned-access)
    add_executable(psp2_gpuprof src/kernel.c src/continuations.c)
    option(GPUPROF_BEGIN_DIAGNOSTIC "Log the first Begin result and failed hardware check" OFF)
    if(GPUPROF_BEGIN_DIAGNOSTIC)
        target_compile_definitions(psp2_gpuprof PRIVATE GPUPROF_BEGIN_DIAGNOSTIC=1)
    endif()
    target_include_directories(psp2_gpuprof PRIVATE include src)
    target_compile_options(psp2_gpuprof PRIVATE -Wall -Wextra -Werror
        -mgeneral-regs-only -mno-unaligned-access)
    target_link_options(psp2_gpuprof PRIVATE -nostdlib -Wl,-q)
    target_link_libraries(psp2_gpuprof PRIVATE gpuprof_session
        taihenForKernel_stub taihenModuleUtils_stub SceSysclibForDriver_stub
        SceSysmemForDriver_stub SceThreadmgrForDriver_stub SceProcEventForDriver_stub
        SceIofilemgrForDriver_stub)
    # vita_create_self is a macro that mutates its flags; isolate each call.
    function(create_kernel_self)
        vita_create_self(psp2_gpuprof.skprx psp2_gpuprof CONFIG exports.yml UNSAFE)
    endfunction()
    create_kernel_self()
    vita_create_stubs(stubs psp2_gpuprof ${CMAKE_CURRENT_SOURCE_DIR}/exports.yml KERNEL)
    add_custom_target(gpuprof_stubs ALL DEPENDS libpsp2_gpuprof_stub.a libpsp2_gpuprof_stub_weak.a)
    install(DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/stubs/ DESTINATION lib
        FILES_MATCHING PATTERN "*.a")
    install(FILES include/psp2_gpuprof.h DESTINATION include)
    install(FILES include/psp2_gpuprof_presets.h DESTINATION include)
    install(FILES include/psp2_gpuprof_trace.h DESTINATION include)
    install(TARGETS psp2_gpuprof_client ARCHIVE DESTINATION lib)
    add_library(gpuprof_capture_example OBJECT examples/capture.c)
    target_include_directories(gpuprof_capture_example PRIVATE include)
    target_compile_options(gpuprof_capture_example PRIVATE -Wall -Wextra -Werror)
    add_library(gpuprof_trace_example OBJECT examples/trace.c)
    target_include_directories(gpuprof_trace_example PRIVATE include)
    target_compile_options(gpuprof_trace_example PRIVATE -Wall -Wextra -Werror)
    add_executable(gpuprof_link_check examples/link_check.c)
    target_include_directories(gpuprof_link_check PRIVATE include)
    add_dependencies(gpuprof_link_check gpuprof_stubs)
    target_link_libraries(gpuprof_link_check PRIVATE
        psp2_gpuprof_client
        "${CMAKE_CURRENT_BINARY_DIR}/stubs/libpsp2_gpuprof_stub.a")
    function(create_link_check_self)
        vita_create_self(gpuprof_link_check.self gpuprof_link_check)
    endfunction()
    create_link_check_self()
    add_executable(gpuprof_transfer_probe examples/transfer_probe.c)
    target_include_directories(gpuprof_transfer_probe PRIVATE include)
    target_compile_options(gpuprof_transfer_probe PRIVATE -Wall -Wextra -Werror)
    add_dependencies(gpuprof_transfer_probe gpuprof_stubs)
    target_link_libraries(gpuprof_transfer_probe PRIVATE
        "${CMAKE_CURRENT_BINARY_DIR}/stubs/libpsp2_gpuprof_stub.a"
        SceGxm_stub SceSysmem_stub SceKernelThreadMgr_stub)
    function(create_transfer_probe)
        vita_create_self(gpuprof_transfer_probe.self gpuprof_transfer_probe)
        vita_create_vpk(gpuprof_transfer_probe.vpk "GPUPROF01"
            gpuprof_transfer_probe.self VERSION "01.00" NAME "GPU transfer probe")
    endfunction()
    create_transfer_probe()
    function(create_isolation_probe target title name)
        add_executable(${target} examples/isolation_probe.c)
        target_include_directories(${target} PRIVATE include)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
        if(target STREQUAL "gpuprof_signals_only")
            target_compile_definitions(${target} PRIVATE PROBE_SIGNALS=1)
        endif()
        add_dependencies(${target} gpuprof_stubs)
        target_link_libraries(${target} PRIVATE
            "${CMAKE_CURRENT_BINARY_DIR}/stubs/libpsp2_gpuprof_stub.a"
            vita2d SceGxm_stub SceDisplay_stub SceSysmem_stub
            SceKernelThreadMgr_stub SceCommonDialog_stub SceAppMgr_stub
            SceSysmodule_stub m)
        vita_create_self(${target}.self ${target})
        vita_create_vpk(${target}.vpk ${title} ${target}.self VERSION "01.00" NAME ${name})
    endfunction()
    create_isolation_probe(gpuprof_counters_only "GPUPROF02" "GPU counters only")
    create_isolation_probe(gpuprof_signals_only "GPUPROF03" "GPU signals only")
endif()

    target_include_directories(capture_test PRIVATE include)
    target_compile_options(capture_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME capture COMMAND capture_test)
    add_executable(trace_test tests/trace_test.c)
    target_link_libraries(trace_test PRIVATE psp2_gpuprof_client)
    target_compile_options(trace_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME trace COMMAND trace_test)
endif()
