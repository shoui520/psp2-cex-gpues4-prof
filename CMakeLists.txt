    function(create_isolation_probe target title name)
        if(target STREQUAL "gpuprof_lab")
            add_executable(${target} examples/lab/main.c examples/lab/capture.c)
            if(GPUPROF_LAB_TIMING_VALIDATION)
                target_compile_definitions(${target} PRIVATE GPUPROF_LAB_TIMING_VALIDATION=1)
            endif()
            if(GPUPROF_LAB_TIMING_DIAGNOSTIC)
                target_compile_definitions(${target} PRIVATE GPUPROF_LAB_TIMING_DIAGNOSTIC=1)
            endif()
            if(GPUPROF_LAB_SIGNAL_VALIDATION)
                target_compile_definitions(${target} PRIVATE GPUPROF_LAB_SIGNAL_VALIDATION=1)
            endif()
            target_link_libraries(${target} PRIVATE vita2d ScePower_stub SceCtrl_stub ScePgf_stub)
            target_link_options(${target} PRIVATE
                -Wl,--wrap=sceGxmBeginScene -Wl,--wrap=sceGxmEndScene
                -Wl,--wrap=sceGxmDisplayQueueAddEntry
                -Wl,--wrap=sceGxmDraw -Wl,--wrap=sceGxmSetVertexProgram
                -Wl,--wrap=sceGxmSetFragmentProgram)
        elseif(target STREQUAL "gpuprof_draw_probe")
            add_executable(${target} examples/draw_probe.c)
        elseif(target STREQUAL "gpuprof_work_probe")
            add_executable(${target} examples/work_probe.c)
        elseif(target STREQUAL "gpuprof_timer_probe")
            add_executable(${target} examples/timer_probe.c)
            target_link_libraries(${target} PRIVATE ScePower_stub)
        elseif(target STREQUAL "gpuprof_scenes_probe")
            add_executable(${target} examples/scenes_probe.c)
            target_link_options(${target} PRIVATE
                -Wl,--wrap=sceGxmDraw
                -Wl,--wrap=sceGxmSetVertexProgram
                -Wl,--wrap=sceGxmSetFragmentProgram)
        else()
            add_executable(${target} examples/isolation_probe.c)
        endif()
        target_include_directories(${target} PRIVATE include)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
        if(target STREQUAL "gpuprof_signals_only")
            target_compile_definitions(${target} PRIVATE PROBE_SIGNALS=1)
        endif()
        add_dependencies(${target} gpuprof_stubs)
        target_link_libraries(${target} PRIVATE
            psp2_gpuprof_client
            "${CMAKE_CURRENT_BINARY_DIR}/stubs/libpsp2_gpuprof_stub.a"
            vita2d SceGxm_stub SceDisplay_stub SceSysmem_stub
            SceKernelThreadMgr_stub SceCommonDialog_stub SceAppMgr_stub
            SceSysmodule_stub m)
        vita_create_self(${target}.self ${target})
        if(target STREQUAL "gpuprof_lab")
            vita_create_vpk(${target}.vpk ${title} ${target}.self VERSION "01.00" NAME ${name}
                FILE ${GPUPROF_LAB_SHADERS}/scene_v.gxp shaders/scene_v.gxp
                     ${GPUPROF_LAB_SHADERS}/lit_f.gxp shaders/lit_f.gxp
                     ${GPUPROF_LAB_SHADERS}/heavy_f.gxp shaders/heavy_f.gxp
                     ${GPUPROF_LAB_SHADERS}/warp_f.gxp shaders/warp_f.gxp
                     ${GPUPROF_LAB_SHADERS}/fetch_f.gxp shaders/fetch_f.gxp)
        else()
            vita_create_vpk(${target}.vpk ${title} ${target}.self VERSION "01.00" NAME ${name})
        endif()
    endfunction()
